cmake_minimum_required(VERSION 3.20)
project(Julretsu VERSION 0.3.0 LANGUAGES C CXX)
option(JULRETSU_BUILD_GUI "Build native application" OFF)
option(JULRETSU_ENABLE_LUA "Build Lua integration" OFF)
option(JULRETSU_BUILD_TESTS "Build headless tests" ON)
option(JULRETSU_BUILD_BENCHMARKS "Build deterministic benchmarks" OFF)
option(JULRETSU_ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers on Linux/macOS Clang/GCC" OFF)
if(JULRETSU_BUILD_GUI)
  set(JULRETSU_ENABLE_LUA ON)
endif()
include(cmake/Dependencies.cmake)
function(julretsu_target target)
  set_target_properties(${target} PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(JULRETSU_ENABLE_SANITIZERS)
    if(WIN32 OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      message(FATAL_ERROR "Sanitizers are supported with GCC/Clang on Linux and macOS.")
    endif()
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endfunction()
add_library(julretsu_core src/Types.cpp src/Formula.cpp src/GridCore.cpp)
target_include_directories(julretsu_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
julretsu_target(julretsu_core)
if(JULRETSU_ENABLE_LUA)
  julretsu_prepare_lua_dependencies()
  add_library(julretsu_lua src/LuaEngine.cpp)
  target_link_libraries(julretsu_lua PUBLIC julretsu_core PRIVATE julretsu_sol2 julretsu_lua_runtime)
  julretsu_target(julretsu_lua)
endif()
add_executable(julretsu_headless examples/headless.cpp)
target_link_libraries(julretsu_headless PRIVATE julretsu_core)
julretsu_target(julretsu_headless)
if(JULRETSU_BUILD_TESTS AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests")
  enable_testing()
  add_executable(julretsu_tests tests/core_tests.cpp)
  target_link_libraries(julretsu_tests PRIVATE julretsu_core)
  julretsu_target(julretsu_tests)
  add_test(NAME core_contracts COMMAND julretsu_tests)
  set_tests_properties(core_contracts PROPERTIES TIMEOUT 120)
  add_executable(julretsu_viewport_tests tests/viewport_tests.cpp)
  target_link_libraries(julretsu_viewport_tests PRIVATE julretsu_core)
  julretsu_target(julretsu_viewport_tests)
  add_test(NAME viewport_contracts COMMAND julretsu_viewport_tests)
  if(JULRETSU_ENABLE_LUA)
    add_executable(julretsu_lua_tests tests/lua_tests.cpp)
    target_link_libraries(julretsu_lua_tests PRIVATE julretsu_lua)
    julretsu_target(julretsu_lua_tests)
    add_test(NAME lua_contracts COMMAND julretsu_lua_tests)
    set_tests_properties(lua_contracts PROPERTIES TIMEOUT 60)
  endif()
endif()
if(JULRETSU_BUILD_GUI)
  julretsu_prepare_gui_dependencies()
  add_library(julretsu_ui src/GridUI.cpp)
  target_link_libraries(julretsu_ui PUBLIC julretsu_lua julretsu_imgui)
  julretsu_target(julretsu_ui)
  add_executable(julretsu src/main.cpp src/AllocationMetrics.cpp src/Branding.cpp)
  target_include_directories(julretsu SYSTEM PRIVATE ${stb_SOURCE_DIR})
  add_custom_target(julretsu_assets ALL
    COMMAND ${CMAKE_COMMAND} -E copy_directory
      "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:julretsu>/assets")
  add_dependencies(julretsu julretsu_assets)
  target_link_libraries(julretsu PRIVATE julretsu_ui)
  if(WIN32)
    enable_language(RC)
    configure_file(cmake/AppIcon.rc.in ${CMAKE_CURRENT_BINARY_DIR}/AppIcon.rc @ONLY)
    set_source_files_properties(${CMAKE_CURRENT_BINARY_DIR}/AppIcon.rc PROPERTIES
      OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/app-icon.ico")
    target_sources(julretsu PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/AppIcon.rc)
    target_link_libraries(julretsu PRIVATE psapi)
    set_target_properties(julretsu PROPERTIES WIN32_EXECUTABLE TRUE)
  endif()
  julretsu_target(julretsu)
endif()
if(JULRETSU_BUILD_BENCHMARKS)
  add_executable(julretsu_benchmarks benchmarks/core_benchmarks.cpp)
  target_link_libraries(julretsu_benchmarks PRIVATE julretsu_core)
  julretsu_target(julretsu_benchmarks)
  target_compile_definitions(julretsu_benchmarks PRIVATE JULRETSU_BUILD_CONFIG="$<CONFIG>")
endif()
